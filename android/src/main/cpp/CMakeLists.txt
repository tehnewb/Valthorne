cmake_minimum_required(VERSION 3.22)
project(valthorne_android LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=4 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)

if(NOT DEFINED VALTHORNE_NATIVE_SOURCES)
    message(FATAL_ERROR "Run prepareAndroidNative before configuring the Android renderer")
endif()
set(BGFX "${VALTHORNE_NATIVE_SOURCES}/bgfx")
set(BX "${VALTHORNE_NATIVE_SOURCES}/bx")
set(BIMG "${VALTHORNE_NATIVE_SOURCES}/bimg")
set(SOURCES "${VALTHORNE_NATIVE_SOURCES}")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/shaderc" shaderc)

set(FONT_SOURCES "${VALTHORNE_NATIVE_SOURCES}")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(YOGA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${VALTHORNE_NATIVE_SOURCES}/yoga/yoga" yoga)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../browser/native/font" fonts)
set(GENERATED "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/generated/platform/cpp")
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
set(ALSOFT_BACKEND_OPENSL ON CACHE BOOL "" FORCE)
set(ALSOFT_REQUIRE_OPENSL ON CACHE BOOL "" FORCE)
add_subdirectory("${VALTHORNE_NATIVE_SOURCES}/openal" openal)
add_library(valthorne_android SHARED native_device.cpp native_commands.cpp "${GENERATED}/runtime.cpp" "${GENERATED}/yoga.cpp" "${BGFX}/examples/common/nanovg/nanovg.cpp" "${BGFX}/examples/common/nanovg/nanovg_bgfx.cpp" "${BGFX}/src/amalgamated.cpp" "${BX}/src/amalgamated.cpp" "${BIMG}/src/image.cpp")
target_compile_features(valthorne_android PRIVATE cxx_std_20)
target_sources(valthorne_android PRIVATE "${GENERATED}/audio.cpp")
target_link_libraries(valthorne_android PRIVATE OpenAL)
target_include_directories(valthorne_android PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${GENERATED}" "${BGFX}/include" "${BGFX}/src" "${BGFX}/3rdparty" "${BGFX}/3rdparty/stb" "${BGFX}/examples/common/nanovg" "${BGFX}/3rdparty/khronos" "${BX}/include" "${BIMG}/include" "${BIMG}/3rdparty" "${BIMG}/3rdparty/stb")
target_compile_definitions(valthorne_android PRIVATE BX_CONFIG_DEBUG=0 BIMG_CONFIG_DECODE_ASTC=0 BGFX_CONFIG_MULTITHREADED=1 BGFX_CONFIG_RENDERER_OPENGL=0 BGFX_CONFIG_RENDERER_OPENGLES=30 BGFX_CONFIG_RENDERER_VULKAN=1 MSDFGEN_DISABLE_VARIABLE_FONTS MSDFGEN_DISABLE_SVG MSDFGEN_DISABLE_PNG)
target_link_libraries(valthorne_android PRIVATE valthorne-fonts yogacore android log EGL GLESv3 dl)
target_link_options(valthorne_android PRIVATE "-Wl,-z,max-page-size=16384")
