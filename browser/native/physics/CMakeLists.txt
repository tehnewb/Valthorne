cmake_minimum_required(VERSION 3.20)
project(ValthorneJolt LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS OFF CACHE BOOL "" FORCE)
set(FLOATING_POINT_EXCEPTIONS_ENABLED OFF CACHE BOOL "" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(ENABLE_OBJECT_STREAM OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(DISABLE_CUSTOM_ALLOCATOR ON CACHE BOOL "" FORCE)
set(OBJECT_LAYER_BITS 32 CACHE STRING "" FORCE)
set(USE_WASM_SIMD OFF CACHE BOOL "" FORCE)
add_subdirectory("${JOLT_SOURCE}/Build" jolt EXCLUDE_FROM_ALL)

add_executable(valthorne-jolt "${JOLT_GENERATED}/glue.cpp")
target_link_libraries(valthorne-jolt PRIVATE Jolt)
target_compile_options(valthorne-jolt PRIVATE -O3 -fno-rtti -fno-exceptions -include "${JOLT_GENERATED}/JoltJS.h")
target_link_options(valthorne-jolt PRIVATE
    -O3 -fno-rtti -fno-exceptions
    "--post-js=${JOLT_GENERATED}/glue.js"
    "--post-js=${JOLT_BINDINGS}/helpers.js"
    -sEXPORT_ES6=1 -sEXPORT_NAME=ValthorneJolt -sMODULARIZE=1
    -sENVIRONMENT=web,node -sFILESYSTEM=0 -sALLOW_TABLE_GROWTH=1
    -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=16777216 -sMAXIMUM_MEMORY=268435456
    -sSTACK_SIZE=1048576 -sABORTING_MALLOC=1
    "-sEXPORTED_RUNTIME_METHODS=HEAP8,HEAPU8,HEAP16,HEAPU16,HEAP32,HEAPU32,HEAPF32,HEAPF64,UTF8ToString")
set_target_properties(valthorne-jolt PROPERTIES SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${JOLT_OUTPUT}")
