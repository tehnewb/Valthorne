cmake_minimum_required(VERSION 3.20)
project(ValthorneBrowserFonts LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FT_DISABLE_ZLIB TRUE CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 TRUE CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG TRUE CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ TRUE CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI TRUE CACHE BOOL "" FORCE)
add_subdirectory("${FONT_SOURCES}/freetype" freetype)
set(HB_BUILD_UTILS OFF CACHE BOOL "" FORCE)
set(HB_BUILD_SUBSET OFF CACHE BOOL "" FORCE)
set(HB_HAVE_FREETYPE ON CACHE BOOL "" FORCE)
add_subdirectory("${FONT_SOURCES}/harfbuzz" harfbuzz)
set(MSDFGEN_CORE_ONLY ON CACHE BOOL "" FORCE)
set(MSDFGEN_BUILD_STANDALONE OFF CACHE BOOL "" FORCE)
set(MSDFGEN_USE_VCPKG OFF CACHE BOOL "" FORCE)
set(MSDFGEN_USE_SKIA OFF CACHE BOOL "" FORCE)
set(MSDFGEN_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${FONT_SOURCES}/msdfgen" msdfgen)
add_library(valthorne-fonts STATIC "${FONT_SOURCES}/msdfgen/ext/import-font.cpp")
target_include_directories(valthorne-fonts PUBLIC "${FONT_SOURCES}/msdfgen")
target_link_libraries(valthorne-fonts PUBLIC freetype harfbuzz msdfgen-core)
target_compile_definitions(valthorne-fonts PRIVATE MSDFGEN_DISABLE_VARIABLE_FONTS MSDFGEN_DISABLE_SVG MSDFGEN_DISABLE_PNG)
